\begin{question}
    On opening day, Master Fund LP has the following economic balance sheet:

    \begin{table}[htbp]
        \centering
        \begin{tabular}{|l|l|}
            \hline
            Cash              & \$100 million  \\
            \hline
            Financial assets  & \$200 million  \\
            \hline
            Debt              & \$180 million  \\
            \hline
            Equity            & \$120 million  \\
            \hline
        \end{tabular}
        \caption{资产负债表简表}
    \end{table}
    Assume Master Fund LP finances a long position in \$100 million worth of an equity at the margin requirement of 30\%. It invests \$30 million of its own funds and borrows \$70 from a commercial bank, immediately after the trade, its margin account has \$30 million of equity and \$70 million of loan. What is the change in the firm's leverage ratio after this trade?
\end{question}

\begin{choices}
    \item From 1.5 to 2.5
    \item From 2.50 to 3.08
    \item From 2.5 to 4.11
    \item From 1.500 to 1.500
\end{choices}

\begin{correctanswer}
    B
\end{correctanswer}

\begin{explanation}
    \textbf{A选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 初始杠杆率计算错误
        \item 交易后杠杆率计算错误
    \end{itemize}

    \textbf{B选项正确。}
    这个说法正确：
    \begin{itemize}
        \item 初始杠杆率 = 资产/权益 = 300/120 = 2.5
        \item 交易后资产 = \$70(现金) + \$300(金融资产) = \$370
        \item 交易后负债 = \$180(债务) + \$70(保证金贷款) = \$250
        \item 交易后权益 = \$370 - \$250 = \$120
        \item 交易后杠杆率 = 370/120 = 3.08
    \end{itemize}

    \textbf{C选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 交易后杠杆率计算错误
        \item 高估了杠杆率变化
    \end{itemize}

    \textbf{D选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 杠杆率保持不变的说法错误
        \item 交易导致杠杆率上升
    \end{itemize}
\end{explanation}

\begin{question}
    An investment bank finds that current market has been highly volatile but its own capital base is large enough. However, the counterparty of this bank asks for higher and higher collateral for the bank’s repo and repo rollover. This phenomenon shows:
\end{question}

\begin{choices}
    \item balance sheet risk
    \item transactions liquidity risk
    \item systematic risk
    \item maturity transformation risk
\end{choices}

\begin{correctanswer}
    A
\end{correctanswer}

\begin{explanation}
    \textbf{A选项正确。}
    这个说法正确：
    \begin{itemize}
        \item 这是典型的资金流动性风险或资产负债表风险
        \item 市场波动或信用状况恶化时，交易对手提高抵押品要求
        \item 即使银行资本充足，也可能面临融资条件收紧
    \end{itemize}

    \textbf{B选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 交易流动性风险指市场买卖难度
        \item 与抵押品要求变化无直接关系
    \end{itemize}

    \textbf{C选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 系统性风险指整个金融系统风险
        \item 本题更侧重个体银行的融资风险
    \end{itemize}

    \textbf{D选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 到期错配风险指短借长贷等结构性风险
        \item 与抵押品要求变化无直接关系
    \end{itemize}
\end{explanation}

The stock's daily volatility is 2.59\%. Allen is interesting in calculating the LVaR of the portfolio, Let's assume the stock's arithmetic returns are normally distributed (aka, normal VaR) and the expected daily return rounds to zero. Which is NEAREST to the position's one-day 99.0\% confident liquidity-adjusted value at risk (LVaR)?
\end{question}

\begin{choices}
    \item \$123,000
    \item \$276,000
    \item \$340,000
    \item \$400,000
\end{choices}

\begin{correctanswer}
    C
\end{correctanswer}

\begin{explanation}
    \textbf{A选项错误。}

    \end{itemize}

\begin{question}
    A national regulator is developing a set of indicators to serve as warning signs that the liquidity of the stock market could be decreasing. The regulator is particularly concerned about the potential for a "liquidity black hole" to develop which could destabilize the stock market. Which of the following observations would be of greatest concern to the regulator as a signal of a potential liquidity black hole occurring?
\end{question}

\begin{choices}
    \item A majority of large investors increasingly follow a trend-following trading strategy
    \item Portfolio managers sell stocks that have increased in value and reached their price target
    \item Brokerage firms purchase call options as a hedge against the sale of a large number of put options to institutional investors
    \item Bid-ask spreads for less liquid stocks become narrower
\end{choices}

\begin{correctanswer}
    A
\end{correctanswer}

\begin{explanation}
    \textbf{A选项正确。}
    这个说法正确：
    \begin{itemize}
        \item 大量投资者采用趋势跟踪策略
        \item 可能导致羊群效应
        \item 加剧市场波动和流动性枯竭
    \end{itemize}

    \textbf{B选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 这是正常的投资行为
        \item 不会导致流动性黑洞
    \end{itemize}

    \textbf{C选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 这是正常的对冲行为
        \item 有助于管理风险
    \end{itemize}

    \textbf{D选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 价差收窄表明流动性改善
        \item 不是流动性黑洞的信号
    \end{itemize}
\end{explanation}

\begin{question}
    已知：ROE=18\%，ROA=12\%，Asset=600，Equity=300，Liability=300。
    问，如果要把ROE降低到15\%，应该怎么做？
\end{question}

\begin{choices}
    \item Asset and Liability 同时增加 300
    \item Asset and Liability 同时减少 300
    \item Asset and Liability 同时增加 150
    \item Asset and Liability 同时减少 150
\end{choices}

\begin{correctanswer}
    D
\end{correctanswer}

\begin{explanation}
    \textbf{A选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 资产和负债增加300
        \item 杠杆率增加，ROE会上升
    \end{itemize}

    \textbf{B选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 资产和负债减少300
        \item 杠杆率增加，ROE会上升
    \end{itemize}

    \textbf{C选项错误。}
    这个说法不准确：
    \begin{itemize}
        \item 资产和负债增加150
        \item 杠杆率增加，ROE会上升
    \end{itemize}

    \textbf{D选项正确。}
    这个说法正确：
    \begin{itemize}
        \item 目标ROE = 15\%
        \item 目标净利润 = 300 × 15\% = 45
        \item 资产和负债同时减少150
        \item 保持权益不变，降低杠杆率
        \item 从而降低ROE
    \end{itemize}
\end{explanation}
